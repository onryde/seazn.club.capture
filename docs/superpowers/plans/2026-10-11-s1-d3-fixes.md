> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. Batches 8 Step B onward and Batch 9 need the owner's hands: they run in the main session, never in a subagent.

**Goal:** Fix every D3 finding the owner ruled on ("all", R1): the rejoin after a hold runs out (A4, N29), the square preview (A1, N30), Forget on the notification (B9), Stream settings from Home with the battery row (B6, B8), Send record to Seazn behind its switch (B7), the French tile titles (A2, N31), and the start with no video off the Live Stream view (A3, N28), and file N27–N34.

```kotlin
 * The rejoin reads Reconnecting with cause hold-expired, backs off 2, 4, 8, 16, then 30 s, and never ends itself.

class SessionMachineRejoinTest {
  private val sevenSecondHolds = Configs.descriptor(holdWindowSeconds = mapOf(Transport.SRT to 7, Transport.RTMPS to 7))

  /** Live on sess_42, dropped, and the 7 s hold run out: Paired, the broadcast marked lapsed. Returns its live time. */
  private fun lapsed(rig: MachineRig): Long {
    rig.live()
    val since = rig.session.liveSinceEpochMs!!
    rig.send(Input.Dropped(rig.attempt!!, DropReason.ENDPOINT_CLOSED, null))
```

```kotlin
data class PreviewSizeReport(
  val listed: Int,
  val encodeListed: Boolean,
  val sameArea: Int,
  /** The size the preview gets: ours when a 16:9 size is listed, else StreamPack's. 0 x 0 when nothing is listed. */
  val pickWidth: Int,
  val pickHeight: Int,
  /** N30: whether that size has the encode's aspect, 16:9. */
  val pickAspect16x9: Boolean,
  /** `ours` (the 16:9 rule) or `vendor` (StreamPack's area rule, when no 16:9 size is listed). */
  val pickSource: String,
)

/**
 * The preview's size (N30, spec §4). StreamPack 3.2.1's own rule ([vendorPick], read by javap: `CameraSizes.getPreviewOutputSize`,
 * then `SizeExtensionsKt.closestTo`) sorts by area ascending (stable), reverses, and takes the FIRST size whose area is
 * nearest the view's, so an equal-area square (960 x 960 against 1280 x 720) can win. Ours ([pick16x9]) takes the
 * encode's aspect first: among the listed 16:9 sizes, the nearest in area to 1280 x 720, the first listed on a tie. With
 * none listed, StreamPack's pick stands and the record says `vendor`.
 */
object PreviewSizes {
  private val TARGET = Encode.WIDTH * Encode.HEIGHT

  fun vendorPick(sizes: List<Pair<Int, Int>>): Pair<Int, Int>? {
    var pick: Pair<Int, Int>? = null
    var least = 0
    for (size in sizes.sortedBy { it.first * it.second }.reversed()) {
      val apart = abs(size.first * size.second - TARGET)
      if (pick == null || least > apart) {
        pick = size
        least = apart
      }
    }
    return pick
  }

  fun pick16x9(sizes: List<Pair<Int, Int>>): Pair<Int, Int>? =
    sizes.filter { it.first * 9 == it.second * 16 }.minByOrNull { abs(it.first * it.second - TARGET) }

  fun report(sizes: List<Pair<Int, Int>>): PreviewSizeReport {
    val ours = pick16x9(sizes)
    val pick = ours ?: vendorPick(sizes)
    val encode = Encode.WIDTH to Encode.HEIGHT
    return PreviewSizeReport(
      listed = sizes.size,
      encodeListed = encode in sizes,
      sameArea = sizes.count { it != encode && it.first * it.second == TARGET },
      pickWidth = pick?.first ?: 0,
      pickHeight = pick?.second ?: 0,
      pickAspect16x9 = ours != null,
      pickSource = if (ours != null) "ours" else "vendor",
    )
  }

  /** The record line: numbers, one boolean and one plain word, all on the allow-list. */
  fun entry(report: PreviewSizeReport): RecordEntry =
    RecordEntry(
      "preview-size",
      listOf(
        "listed" to report.listed,
        "encodeListed" to report.encodeListed,
        "sameArea" to report.sameArea,
        "pickWidth" to report.pickWidth,
        "pickHeight" to report.pickHeight,
        "pickAspect16x9" to report.pickAspect16x9,
        "pickSource" to report.pickSource,
      ),
    )
}
```

```ts
  readonly openSettings: () => Promise<void>;

/** R5 (P9): the battery state from native, the page by React Native's own call. Never throws. */
export function createNativePower({ native, os, openSettings, logger }: Deps): PowerPort {
```

```ts
/**
 * R4 (owner, 2026-10-11): what Send record authenticates with — the last stream code's token, kept while its code is
 * saved and for 24 h after the code is dropped (Forget, Replaced, Taken, a Stop), for uploads only. Pure.
 */
export type UploadCredential = {
  /** The code's URL (`SavedStream.codeUrl`); the upload is its `/record`. */
  readonly codeUrl: string;
  /** The code's Bearer. A secret: never log it, never put it in a record line or a snapshot. */
  readonly token: string;
  /** Null while the code is saved; the drop time plus [KEEP_AFTER_DROP_MS] once dropped. */
  readonly expiresAtMs: number | null;
};

export const KEEP_AFTER_DROP_MS = 24 * 60 * 60 * 1000;

export type CredentialStep =
  | { readonly kind: 'keep' }
  | { readonly kind: 'write'; readonly credential: UploadCredential }
  | { readonly kind: 'delete' };

type Saved = { readonly codeUrl: string; readonly token: string };

const KEEP: CredentialStep = { kind: 'keep' };

/** One decision from what is stored, the code saved now (or null), and the clock. */
export function nextCredential(
  stored: UploadCredential | null,
  saved: Saved | null,
  nowMs: number,
): CredentialStep {
  if (saved !== null) {
    const fresh: UploadCredential = {
      codeUrl: saved.codeUrl,
      token: saved.token,
      expiresAtMs: null,
    };
    const same =
      stored !== null &&
      stored.codeUrl === fresh.codeUrl &&
      stored.token === fresh.token &&
      stored.expiresAtMs === null;
    return same ? KEEP : { kind: 'write', credential: fresh };
  }
  if (stored === null) return KEEP;
  if (stored.expiresAtMs === null)
    return { kind: 'write', credential: { ...stored, expiresAtMs: nowMs + KEEP_AFTER_DROP_MS } };
  return nowMs >= stored.expiresAtMs ? { kind: 'delete' } : KEEP;
}

/** The credential an upload may use at [nowMs], or null. */
export function usableCredential(
  stored: UploadCredential | null,
  nowMs: number,
): UploadCredential | null {
  if (stored === null) return null;
  return stored.expiresAtMs !== null && nowMs >= stored.expiresAtMs ? null : stored;
}

const VERSION = 1;

export function encodeCredential(credential: UploadCredential): string {
  return JSON.stringify({ v: VERSION, ...credential });
}

/** The anti-corruption read of what the store holds: anything not exactly this shape is no credential. */
export function decodeCredential(text: string): UploadCredential | null {
  let data: unknown;
  try {
    data = JSON.parse(text);
  } catch {
    return null;
  }
  if (typeof data !== 'object' || data === null) return null;
  const { v, codeUrl, token, expiresAtMs } = data as Record<string, unknown>;
  if (v !== VERSION || typeof codeUrl !== 'string' || typeof token !== 'string') return null;
  if (expiresAtMs !== null && typeof expiresAtMs !== 'number') return null;
  return { codeUrl, token, expiresAtMs };
}
```

```ts
import { parseStreamCodeText } from '@/domain/credentials/streamCode';
import {
  decodeCredential,
  encodeCredential,
  nextCredential,
  usableCredential,
  type UploadCredential,
} from '@/domain/credentials/uploadCredential';
import type { KeyValueStore } from '@/services/KeyValueStore';
import type { Logger } from '@/services/logger';
import type { ModeStore } from '@/services/modeStore';

export const UPLOAD_CREDENTIAL_KEY = 'upload-credential.v1';

/** R4: the credential Send record may use now. Kept in secure storage behind the key-value port; never in a record. */
export interface UploadCredentialStore {
  /** The usable credential, or null (none, expired, unreadable, or the switch off). Never rejects. */
  current(): Promise<UploadCredential | null>;
}

type Deps = {
  readonly kv: KeyValueStore;
  readonly modeStore: ModeStore;
  readonly clock: () => Date;
  readonly logger: Logger;
  /** P10: `FEATURES.recordSend`, passed in by the composition root. Off, nothing is ever stored. */
  readonly enabled: boolean;
};

/**
 * Follows the saved stream code (R4): renewed at every save, its 24 h started when the code goes, deleted after them.
 * Writes run one at a time, so a burst of store publishes cannot interleave a set and a delete.
 */
export function createUploadCredentialStore({
  kv,
  modeStore,
  clock,
  logger,
  enabled,
}: Deps): UploadCredentialStore {
  if (!enabled) return { current: async () => null };
  let queue: Promise<void> = Promise.resolve();
  const run = (work: () => Promise<void>) => {
    queue = queue
      .then(work)
      .catch(() => logger.warn('store.write-refused', { action: 'upload-credential' }));
    return queue;
  };
  const read = async (): Promise<UploadCredential | null> => {
    try {
      const text = await kv.get(UPLOAD_CREDENTIAL_KEY);
      return text === null ? null : decodeCredential(text);
    } catch {
      return null;
    }
  };
  const savedNow = () => {
    const snapshot = modeStore.getSnapshot();
    if (snapshot.status !== 'ready') return undefined;
    const code = snapshot.saved.codes.stream;
    if (code === undefined || code.stream === null) return null;
    const parsed = parseStreamCodeText(code.raw);
    return parsed.ok ? { codeUrl: code.stream.codeUrl, token: parsed.value.token } : null;
  };
  const follow = () => {
    const saved = savedNow();
    if (saved === undefined) return;
    void run(async () => {
      const step = nextCredential(await read(), saved, clock().getTime());
      if (step.kind === 'write')
        await kv.set(UPLOAD_CREDENTIAL_KEY, encodeCredential(step.credential));
      if (step.kind === 'delete') await kv.delete(UPLOAD_CREDENTIAL_KEY);
    });
  };
  modeStore.subscribe(follow);
  follow();
  return {
    current: async () => {
      await queue;
      const stored = await read();
      const usable = usableCredential(stored, clock().getTime());
      if (stored !== null && usable === null) void run(() => kv.delete(UPLOAD_CREDENTIAL_KEY));
      return usable;
    },
  };
}
```

```ts
import type { Logger } from '@/services/logger';
import { type LogFields, type LogValue, scrubEvent, scrubFields } from '@/services/scrub';
import { parseRecordLine } from '@/services/sessionRecord';

/**
 * Provisional (P4): these two shapes are the web brief's §1 as asked, not a vendored contract. When the web publishes
 * its record contract, `pnpm contracts:vendor` and the pin test replace them, and nothing else in the app changes.
 */
type RecordUploadHeaders = {
  readonly Authorization: string;
  readonly 'Content-Type': 'application/x-ndjson';
};
type RecordUploadAnswer = { readonly ref: string };

export type RecordSendResult =
  'sent' | 'refused-local' | 'unauthorised' | 'too-large' | 'rate-limited' | 'failed';
export type RecordSendOutcome = {
  readonly result: RecordSendResult;
  /** The server's short reference, for the operator to read to the organiser. Never logged. */
  readonly ref: string | null;
  readonly status: number | null;
  readonly bytes: number;
};

/** R3: one request with an answer, on the operator's tap; it never touches the engine bridge (spec §7). */
export interface RecordUploadPort {
  send(lines: readonly string[], credential: UploadCredential): Promise<RecordSendOutcome>;
}

export type PostFetch = (
  url: string,
  init: { method: 'POST'; headers: RecordUploadHeaders; body: string; signal: AbortSignal },
) => Promise<{ status: number; json(): Promise<unknown> }>;

export const RECORD_MAX_LINES = 1000;
export const RECORD_MAX_BYTES = 1_000_000;
export const RECORD_TIMEOUT_MS = 15_000;
const REF = /^[A-Za-z0-9-]{1,16}$/;

const STATUS: Readonly<Record<number, RecordSendResult>> = {
  401: 'unauthorised',
  413: 'too-large',
  429: 'rate-limited',
};

const isLogValue = (value: unknown): value is LogValue =>
  value === null || ['string', 'number', 'boolean'].includes(typeof value);

/**
 * Fail closed (spec §7): every line must be a record line the scrub would leave exactly as it is, with the token held
 * as a secret. One line that names the token, does not parse, or would change under the scrub sends nothing.
 */
export function linesAreSafe(lines: readonly string[], token: string): boolean {
  const held = { secrets: [token], streamIds: [] };
  return lines.every((line) => {
    if (line.includes(token)) return false;
    try {
      const entry = parseRecordLine(line);
      if (scrubEvent(entry.event, held) !== entry.event) return false;
      if (!Object.values(entry.fields).every(isLogValue)) return false;
      const fields = entry.fields as LogFields;
      const scrubbed = scrubFields(fields, held, true);
      return Object.keys(fields).every((key) => scrubbed[key] === fields[key]);
    } catch {
      return false;
    }
  });
}

export function createRecordUploadPort({
  fetch,
  logger,
}: {
  fetch: PostFetch;
  logger: Logger;
}): RecordUploadPort {
  const outcome = (
    result: RecordSendResult,
    status: number | null,
    bytes: number,
    ref: string | null = null,
  ): RecordSendOutcome => {
    logger.info('record.send', { result, status, bytes });
    return { result, ref, status, bytes };
  };
  return {
    send: async (all, credential) => {
      const lines = all.slice(-RECORD_MAX_LINES);
      if (lines.length === 0) return outcome('failed', null, 0);
      if (!linesAreSafe(lines, credential.token)) return outcome('refused-local', null, 0);
      const body = `${lines.join('\n')}\n`;
      const bytes = new TextEncoder().encode(body).length;
      if (bytes > RECORD_MAX_BYTES) return outcome('too-large', null, bytes);
      const abort = new AbortController();
      const timer = setTimeout(() => abort.abort(), RECORD_TIMEOUT_MS);
      try {
        const answer = await Promise.race([
          fetch(`${credential.codeUrl}/record`, {
            method: 'POST',
            headers: {
              Authorization: `Bearer ${credential.token}`,
              'Content-Type': 'application/x-ndjson',
            },
            body,
            signal: abort.signal,
          }),
          new Promise<never>((_, reject) =>
            abort.signal.addEventListener('abort', () => reject(new Error('timeout'))),
          ),
        ]);
        if (answer.status !== 201)
          return outcome(STATUS[answer.status] ?? 'failed', answer.status, bytes);
        const ref = ((await answer.json()) as Partial<RecordUploadAnswer> | null)?.ref;
        return typeof ref === 'string' && REF.test(ref)
          ? outcome('sent', 201, bytes, ref)
          : outcome('failed', 201, bytes);
      } catch {
        return outcome('failed', null, bytes);
      } finally {
        clearTimeout(timer);
      }
    },
  };
}
```

```ts
import { useCallback, useEffect, useMemo, useRef, useState } from 'react';
import { usePorts } from '@/hooks/usePorts';

export type SendState =
  | { readonly kind: 'idle' }
  | { readonly kind: 'sending' }
  | { readonly kind: 'sent'; readonly ref: string }
  | { readonly kind: 'failed' };

const IDLE: SendState = { kind: 'idle' };
const SENDING: SendState = { kind: 'sending' };
const FAILED: SendState = { kind: 'failed' };

/**
 * R3: Send record to Seazn, on the operator's tap only. What it sends is what Share sends (R9: the JS ring). One
 * request at a time: a tap while one is in flight is nothing. The credential is read at mount and at every return to
 * the foreground (it can expire in a pocket).
 */
export function useRecordSend(): {
  readonly state: SendState;
  readonly available: boolean;
  send(): void;
} {
  const { uploadCredential, recordUpload, record, foreground } = usePorts();
  const [available, setAvailable] = useState(false);
  const [state, setState] = useState<SendState>(IDLE);
  const inFlight = useRef(false);
  const alive = useRef(true);
  useEffect(() => {
    alive.current = true;
    const read = () => {
      void uploadCredential
        .current()
        .then((credential) => alive.current && setAvailable(credential !== null));
    };
    read();
    const off = foreground.subscribe(read);
    return () => {
      alive.current = false;
      off();
    };
  }, [uploadCredential, foreground]);
  const send = useCallback(() => {
    if (inFlight.current) return;
    inFlight.current = true;
    setState(SENDING);
    void (async () => {
      const credential = await uploadCredential.current();
      const outcome =
        credential === null ? null : await recordUpload.send(record.lines(), credential);
      inFlight.current = false;
      if (!alive.current) return;
      setState(
        outcome !== null && outcome.result === 'sent' && outcome.ref !== null
          ? { kind: 'sent', ref: outcome.ref }
          : FAILED,
      );
    })();
  }, [uploadCredential, recordUpload, record]);
  return useMemo(() => ({ state, available, send }), [state, available, send]);
}
```
